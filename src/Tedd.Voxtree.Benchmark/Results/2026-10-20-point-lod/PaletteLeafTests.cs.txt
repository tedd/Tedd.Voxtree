    static uint[] Values(uint maximum=uint.MaxValue)
    {
        var a=Enumerable.Range(0,512).Select(i=>(uint)(i&1)).ToArray();
        for(int x=0;x<2;x++)for(int y=0;y<2;y++)for(int z=0;z<2;z++)a[(x*8+y)*8+z]=7;
        a[123]=maximum;return a;
    }
    static int Lod(uint[] a,int x,int y,int z,int levels)
    {
        int side=1<<levels;uint v=a[(x*side+y)*side+z];
        for(int lod=levels;lod>0;lod--){int n=1<<lod,ax=x&-n,ay=y&-n,az=z&-n;bool same=true;
            for(int xx=ax;xx<ax+n&&same;xx++)for(int yy=ay;yy<ay+n&&same;yy++)for(int zz=az;zz<az+n;zz++)if(a[(xx*side+yy)*side+zz]!=v){same=false;break;}
            if(same)return lod;
        }return 0;
    }
    static byte[] Reference(uint[] values,int levels,bool narrow,bool generic)
    {
        int side=1<<levels,h=generic?3:2,w=narrow?4:8;
        var keys=Enumerable.Range(0,values.Length).Select(i=>(Value:values[i],Lod:Lod(values,i/(side*side),i/side%side,i%side,levels))).ToArray();
        var states=keys.Distinct().OrderBy(x=>x.Value).ThenBy(x=>x.Lod).ToArray();Assert.InRange(states.Length,1,256);
        var ids=states.Select((s,i)=>(s,i)).ToDictionary(x=>x.s,x=>(byte)x.i);
        var bytes=new byte[h+2+values.Length+states.Length*w];bytes[0]=0x51;bytes[1]=(byte)(0xc3|(levels<<2));if(generic)bytes[2]=(byte)(narrow?1:2);
        BinaryPrimitives.WriteUInt16LittleEndian(bytes.AsSpan(h),checked((ushort)states.Length));
        for(int i=0;i<keys.Length;i++)bytes[h+2+i]=ids[keys[i]];
        for(int i=0;i<states.Length;i++){
            ulong word=states[i].Value|((ulong)states[i].Lod<<(narrow?16:32));
            if(narrow)BinaryPrimitives.WriteUInt32LittleEndian(bytes.AsSpan(h+2+values.Length+i*w),(uint)word);
            else BinaryPrimitives.WriteUInt64LittleEndian(bytes.AsSpan(h+2+values.Length+i*w),word);
        }return bytes;
    }
    [Fact]
    public void AllThreeApisEmitIndependentCanonicalBytesAndPreserveFullValues()
    {
        Assert.Equal(1,Octree.FormatVersion);Assert.Equal(2,Octree<ushort>.FormatVersion);
        Assert.Equal(3,Octree.PaletteLeafFormatVersion);Assert.Equal(3,Octree<ushort>.PaletteLeafFormatVersion);
        var values=Values();var small=Values(ushort.MaxValue);var a=new Octree(3,values);var b=new Octree<ushort>(3,small.Select(v=>(ushort)v).ToArray());var c=new Octree<uint>(3,values);
        Assert.Equal(Reference(values,3,false,false),a.Data.ToArray());Assert.Equal(Reference(small,3,true,true),b.Data.ToArray());Assert.Equal(Reference(values,3,false,true),c.Data.ToArray());
        Assert.True(a.AsSpan().IsWellFormed());Assert.True(b.AsSpan().IsWellFormed());Assert.True(c.AsSpan().IsWellFormed());
        var lookup=a.CreateLookup();var restored=Octree.FromEncoded(a.Data.Span);
        for(int i=0;i<512;i++){
            int x=i/64,y=i/8%8,z=i%8,expected=Lod(values,x,y,z,3);
            Assert.Equal(values[i],a.Get(x,y,z,out int al));Assert.Equal(expected,al);Assert.Equal(values[i],lookup.Get(x,y,z,out al));Assert.Equal(expected,al);
            Assert.Equal(values[i],restored.Get(x,y,z,out al));Assert.Equal(expected,al);Assert.Equal(values[i],c.Get(x,y,z,out al));Assert.Equal(expected,al);
            Assert.Equal((ushort)small[i],b.Get(x,y,z,out al));Assert.Equal(Lod(small,x,y,z,3),al);
        }
        var output=new uint[512];a.CopyTo(output);Assert.Equal(values,output);
        var narrow=new ushort[512];b.CopyTo(narrow);Assert.Equal(small.Select(v=>(ushort)v),narrow);
    }
    [Fact]
    public void ExactCapacityMortonAndSnapshotOwnershipRemainValid()
    {
        var values=Values();var a=new Octree(3,values);int required=Octree.GetRequiredSize(values,3);
        Assert.Equal(a.EncodedLength,required);Assert.True(required<=Octree.GetMaximumSize(3));
        var bytes=Enumerable.Repeat((byte)0xa5,required+7).ToArray();Assert.True(Octree.TryBuild(values,3,bytes,out int written));Assert.Equal(required,written);Assert.Equal(a.Data.ToArray(),bytes[..written]);Assert.All(bytes[written..],v=>Assert.Equal((byte)0xa5,v));
        var shortBytes=new byte[required-1];Assert.False(Octree.TryBuild(values,3,shortBytes,out written));Assert.Equal(0,written);
        var morton=new uint[512];DenseVoxel.Convert(values,morton,3,DenseVoxelLayout.Linear,DenseVoxelLayout.Morton);
        var b=new Octree(3);b.Build(morton,DenseVoxelLayout.Morton);Assert.Equal(a.Data.ToArray(),b.Data.ToArray());
        var decoded=new uint[512];a.CopyTo(decoded,DenseVoxelLayout.Morton);Assert.Equal(morton,decoded);
        var backing=new byte[required+13];a.Data.Span.CopyTo(backing.AsSpan(7));var view=new OctreeSpan(backing.AsSpan(7,required));var copy=view.CreateLookup();var owned=a.CreateLookup();
        Array.Clear(backing);a.BuildUniform(0);Parallel.For(0,512,i=>{Assert.Equal(values[i],copy.Get(i/64,i/8%8,i%8));Assert.Equal(values[i],owned.Get(i/64,i/8%8,i%8));});
    }
    [Fact]
    public void PaletteCount256IsNotAnEmptySentinelAnd257StatesFallback()
    {
        var values=Enumerable.Range(0,4096).Select(i=>(uint)(i%256)).ToArray();var a=new Octree(4,values);var b=new Octree<ushort>(4,values.Select(v=>(ushort)v).ToArray());
        Assert.Equal((byte)0x51,a.Data.Span[0]);Assert.Equal(256,BinaryPrimitives.ReadUInt16LittleEndian(a.Data.Span.Slice(2,2)));Assert.True(a.AsSpan().IsWellFormed());Assert.Equal(255u,a.Get(0,15,15));
        Assert.Equal((byte)0x51,b.Data.Span[0]);Assert.Equal(256,BinaryPrimitives.ReadUInt16LittleEndian(b.Data.Span.Slice(3,2)));Assert.True(b.AsSpan().IsWellFormed());
        var many=Enumerable.Range(0,4096).Select(i=>(uint)(i%257)).ToArray();var fallback=new Octree(4,many);Assert.Equal((byte)0x4f,fallback.Data.Span[0]);Assert.Equal(0,fallback.Data.Span[1]&3);
    }
    [Fact]
    public void HeaderDiscriminatorsAndCountLengthBoundsAreStrict()
    {
        var original=new Octree(3,Values()).Data.ToArray();Assert.Equal((byte)0x51,original[0]);
        foreach(byte magic in new byte[]{0x4f,0x50,0x52}){var data=(byte[])original.Clone();data[0]=magic;Assert.False(OctreeSpan.TryCreate(data,out _));}
        foreach(int count in new[]{0,257}){var data=(byte[])original.Clone();BinaryPrimitives.WriteUInt16LittleEndian(data.AsSpan(2),checked((ushort)count));Assert.False(OctreeSpan.TryCreate(data,out _));}
        foreach(int version in new[]{0,1,2}){var data=(byte[])original.Clone();data[1]=(byte)((data[1]&63)|(version<<6));Assert.False(OctreeSpan.TryCreate(data,out _));}
        foreach(int kind in new[]{0,1,2}){var data=(byte[])original.Clone();data[1]=(byte)((data[1]&252)|kind);Assert.False(OctreeSpan.TryCreate(data,out _));}
        Assert.False(OctreeSpan.TryCreate(original[..^1],out _));Assert.False(OctreeSpan.TryCreate(original.Concat(new byte[]{0}).ToArray(),out _));
        var typed=new Octree<ushort>(3,Values(ushort.MaxValue).Select(v=>(ushort)v).ToArray()).Data.ToArray();Assert.False(OctreeSpan.TryCreate(typed,out _));Assert.False(OctreeSpan<ushort>.TryCreate(original,out _));typed[2]=2;Assert.False(OctreeSpan<ushort>.TryCreate(typed,out _));
    }
    [Fact]
    public void InvalidIndexesAndUnsignedLodsFailSafelyIncludingBorrowedMutation()
    {
        var original=new Octree(3,Values()).Data.ToArray();int table=2+2+512;
        var bad=(byte[])original.Clone();var view=new OctreeSpan(bad);bad[4]=255;
        Assert.False(view.TryGet(0,0,0,out uint value,out int lod));Assert.Equal(0u,value);Assert.Equal(0,lod);Assert.False(view.IsWellFormed());
        var untouched=Enumerable.Repeat(99u,512).ToArray();Assert.False(view.TryCopyTo(untouched));Assert.All(untouched,v=>Assert.Equal(99u,v));
        foreach(uint high in new[]{0x80000000u,uint.MaxValue}){
            var data=(byte[])original.Clone();int code=data[4];BinaryPrimitives.WriteUInt64LittleEndian(data.AsSpan(table+8*code),((ulong)high<<32)|7u);
            Assert.True(OctreeSpan.TryCreate(data,out var current));Assert.False(current.IsWellFormed());Assert.False(current.TryGet(0,0,0,out value,out lod));Assert.Equal(0u,value);Assert.Equal(0,lod);Assert.Throws<FormatException>(()=>Octree.FromEncoded(data));
        }
    }
    [Fact]
    public void DuplicateUnusedUnsortedAndNonmaximalPartitionsAreRejected()
    {
        var original=new Octree(3,Values()).Data.ToArray();int table=516;
        var duplicate=(byte[])original.Clone();duplicate.AsSpan(table,8).CopyTo(duplicate.AsSpan(table+8,8));Assert.False(new OctreeSpan(duplicate).IsWellFormed());
        var unused=(byte[])original.Clone();byte removed=unused[4];for(int i=4;i<table;i++)if(unused[i]==removed)unused[i]=(byte)(removed==0?1:0);Assert.False(new OctreeSpan(unused).IsWellFormed());
        var nonmaximal=new byte[4+512+8];nonmaximal[0]=0x51;nonmaximal[1]=0xcf;nonmaximal[2]=1;BinaryPrimitives.WriteUInt64LittleEndian(nonmaximal.AsSpan(516),7);Assert.True(OctreeSpan.TryCreate(nonmaximal,out var view));Assert.False(view.IsWellFormed());
        BinaryPrimitives.WriteUInt64LittleEndian(nonmaximal.AsSpan(516),7UL|(3UL<<32));Assert.True(new OctreeSpan(nonmaximal).IsWellFormed());
    }
    [Fact]
    public void QueryFiltersNearestAndClippingUseFullRawValues()
    {
        var values=Values();var tree=new Octree(3,values);var box=new VoxelBox(0,0,0,8,8,8);
        foreach(uint wanted in new[]{0u,1u,7u,uint.MaxValue,65535u}){
            int count=values.Count(v=>v==wanted);var filter=VoxelFilter.EqualTo(wanted);
            Assert.Equal(count,tree.CountMatches(box,filter));Assert.Equal(count!=0,tree.Any(box,filter));var hits=new VoxelHit[count];Assert.True(tree.Query(box,filter,hits,out int written));Assert.Equal(count,written);Assert.All(hits,h=>Assert.Equal(wanted,h.Value));
            var expected=Enumerable.Range(0,512).Where(i=>values[i]==wanted).Select(i=>(X:i/64,Y:i/8%8,Z:i%8)).Where(p=>p.X*p.X+p.Y*p.Y+p.Z*p.Z<=9).OrderBy(p=>p.X*p.X+p.Y*p.Y+p.Z*p.Z).ThenBy(p=>p.X).ThenBy(p=>p.Y).ThenBy(p=>p.Z).ToArray();
            Assert.Equal(expected.Length!=0,tree.TryFindNearest(0,0,0,3,filter,out var found));if(expected.Length!=0)Assert.Equal(expected[0],(found.X,found.Y,found.Z));
        }
        var clipped=new VoxelBox(1,2,3,7,6,5);var expectedCopy=new List<uint>();for(int x=1;x<7;x++)for(int y=2;y<6;y++)for(int z=3;z<5;z++)expectedCopy.Add(values[(x*8+y)*8+z]);var output=new uint[clipped.Count];tree.CopyRegionTo(clipped,output);Assert.Equal(expectedCopy,output);
        var invalid=tree.Data.ToArray();invalid[4]=255;Assert.Throws<FormatException>(()=>new OctreeSpan(invalid).CountMatches(box,VoxelFilter.EqualTo(999u)));
    }

        byte[] tree={0x4f,0x44,255,0,1,2,3,4,5,6,7,9};var released=Octree.FromEncoded(tree);for(int i=0;i<8;i++){Assert.Equal((uint)i,released.Get(i/4,i/2%2,i%2,out lod));Assert.Equal(0,lod);}
    }
