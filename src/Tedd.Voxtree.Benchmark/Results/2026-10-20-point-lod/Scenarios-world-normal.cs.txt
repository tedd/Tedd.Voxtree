public static class WorldNormalMode{public static bool Validation;}

public sealed class WorldUInt32 : ISynchronousScenario{
 private readonly string op,pattern;
 private readonly int[] points=new int[1024];private readonly long[] prefix=new long[1025];
 private WorldEntity world=null!;private bool initialized,primed;private int position;private long sum,completed;
 public WorldUInt32(string op,string pattern){this.op=op;this.pattern=pattern;}
 public ScenarioDescriptor Describe()=>new($"normal-world-UInt32-{op}-{pattern}","22",
  Wire.Hash(new{op,pattern,width="UInt32",channels=2,side=1,chunks=12,lod=3,points=1024,affinity=4,primeMs=1000,callerShape="single-library-separate-operation-methods",consumer="exact-prefix-values-plus-hit"}),
  ResetLevel.LogicalState,ThreadAffinity.Any,false,op=="topology"?"One complete topology level search":"One complete public world lookup",["baseline","candidate"]);
 private void Initialize(){if(initialized)return;world=new WorldEntity(1,2);
  for(int i=0;i<12;i++){
   var values=new uint[]{(uint)(i+1),(uint)(i+101)};var chunk=OctreeChunk.FromDense(0,2,values);
   if(op!="topology")world.SetChunk(i-6,0,0,chunk);world.SetLodChunk(3,i-6,0,0,chunk);
  }
  for(int j=0;j<points.Length;j++){
   int index=pattern switch{"repeated"=>0,"blocks32"=>(j>>5)&7,"alternating"=>j&1,"cycle10"=>j%10,"cycle12"=>j%12,_=>30};points[j]=index-6;
   prefix[j+1]=prefix[j]+(index>=12?0:65536+(op=="chunk"?2:index+1+((j>>1)&1)*100));
  }initialized=true;
 }
 public void Prepare(ScenarioContext c){if(c.Variant is not("baseline" or "candidate"))throw new Exception("Unknown variant");Initialize();
  if(!WorldNormalMode.Validation&&!primed){var sw=Stopwatch.StartNew();do{ExecuteBatch(8192);}while(sw.ElapsedMilliseconds<1000);primed=true;}
 }
 public void Reset(ResetContext c){position=0;sum=completed=0;}
 public long ExecuteBatch(int count){switch(op){case "get":AfterGet(count);break;case "lod":AfterLod(count);break;case "chunk":AfterChunk(count);break;case "topology":AfterTopology(count);break;default:throw new Exception();}completed+=count;return count;}
 [MethodImpl(MethodImplOptions.NoInlining)] private void AfterGet(int count){for(int i=0;i<count;i++){int j=position++&1023,x=points[j],channel=(j>>1)&1;bool found;uint result;found=world.TryGet(channel,x,0,0,out result);sum=unchecked(sum+(long)result+(found?65536:0));}return;}
[MethodImpl(MethodImplOptions.NoInlining)] private void AfterLod(int count){for(int i=0;i<count;i++){int j=position++&1023,x=points[j],channel=(j>>1)&1;bool found;uint result;found=world.TryGetLod(3,channel,(long)x<<3,0,0,out result);sum=unchecked(sum+(long)result+(found?65536:0));}return;}
[MethodImpl(MethodImplOptions.NoInlining)] private void AfterChunk(int count){for(int i=0;i<count;i++){int j=position++&1023,x=points[j],channel=(j>>1)&1;bool found;uint result;found=world.TryGetChunk(new ChunkAddress(3,x,0,0),out var chunk);result=(uint)(found?chunk!.ChannelCount:0);sum=unchecked(sum+(long)result+(found?65536:0));}return;}
[MethodImpl(MethodImplOptions.NoInlining)] private void AfterTopology(int count){for(int i=0;i<count;i++){int j=position++&1023,x=points[j],channel=(j>>1)&1;bool found;uint result;found=false;result=0;for(int level=0;level<8;level++){if(world.TryGetLod(level,channel,(long)x<<3,0,0,out result)){found=true;break;}}sum=unchecked(sum+(long)result+(found?65536:0));}return;}
 public ValidationResult Validate()=>new(sum==unchecked(completed/1024*prefix[1024]+prefix[(int)(completed%1024)]));
 public void Cleanup(){}
}
