using Xunit;

namespace Tedd.Voxtree.Tests;

public class ArrayStorageTests
{
    [Theory]
    [InlineData(0)]
    [InlineData(3)]
    [InlineData(5)]
    public void BorrowedDenseSlicesAndOwnedRebuildsRetainTheCapturedSnapshot(int levels)
    {
        int side = 1 << levels;
        var values = Enumerable.Range(0, side * side * side).Select(i => unchecked(uint.MaxValue - (uint)i * 2654435761u)).ToArray();
        var owner = new Octree(levels, values);
        Assert.Equal(2, owner.Data.Span[1] & 3);
        var owned = owner.CreateLookup();
        var source = new byte[owner.EncodedLength + 13];
        owner.Data.Span.CopyTo(source.AsSpan(5));
        var borrowed = new OctreeSpan(source.AsSpan(5, owner.EncodedLength)).CreateLookup();
        Array.Clear(source);
        owner.BuildUniform(0);
        Parallel.For(0, values.Length, i =>
        {
            int x = i / (side * side), y = i / side % side, z = i % side;
            Assert.Equal(values[i], owned.Get(x, y, z, out var a));
            Assert.Equal(values[i], borrowed.Get(x, y, z, out var b));
            Assert.Equal(0, a);
            Assert.Equal(0, b);
        });
    }
}
