public class PackedCountVectorCases
{
    static uint[] Values()=>Enumerable.Range(0,32768).Select(i=>(uint)((i*13+i/32*7)&31)).ToArray();
    [Fact]
    public void CroppedRowsCoverVectorBoundariesAndBothHeaderAlignments()
    {
        var values=Values();var small=values.Select(v=>(ushort)v).ToArray();
        var legacy=new Octree(5,values);var narrow=new Octree<ushort>(5,small);var generic=new Octree<uint>(5,values);
        Assert.Equal(3,legacy.Data.Span[1]&3);Assert.Equal(3,narrow.Data.Span[1]&3);Assert.Equal(3,generic.Data.Span[1]&3);
        foreach(int length in new[]{1,7,8,9,15,16,17,23,31})
        foreach(uint wanted in new uint[]{0,7,31,65535}){
            var box=new VoxelBox(0,0,1,4,3,1+length);int expected=0;
            for(int x=0;x<4;x++)for(int y=0;y<3;y++)for(int z=1;z<1+length;z++)if(values[(x*32+y)*32+z]==wanted)expected++;
            Assert.Equal(expected,legacy.CountMatches(box,VoxelFilter.EqualTo(wanted)));
            Assert.Equal(expected,narrow.CountMatches(box,VoxelFilter<ushort>.EqualTo((ushort)wanted)));
            Assert.Equal(expected,generic.CountMatches(box,VoxelFilter<uint>.EqualTo(wanted)));
        }
        var slab=new VoxelBox(3,0,0,7,32,32);
        int nonzero=0;
        for(int x=3;x<7;x++)for(int y=0;y<32;y++)for(int z=0;z<32;z++)if(values[(x*32+y)*32+z]!=0)nonzero++;
        Assert.Equal(nonzero,legacy.CountMatches(slab,VoxelFilter.NonZero));
        Assert.Equal(nonzero,narrow.CountMatches(slab,VoxelFilter<ushort>.NonZero));
        Assert.Equal(nonzero,generic.CountMatches(slab,VoxelFilter<uint>.NonZero));
    }
    [Theory]
    [InlineData(1)]
    [InlineData(16)]
    [InlineData(17)]
    public void BadLodAtVectorStartEndOrTailCannotBeSkippedByImpossiblePredicate(int z)
    {
        var values=Values();var small=new Octree<ushort>(5,values.Select(v=>(ushort)v).ToArray()).Data.ToArray();
        var wide=new Octree<uint>(5,values).Data.ToArray();var legacy=new Octree(5,values).Data.ToArray();
        int index=(1*32+1)*32+z;
        BinaryPrimitives.WriteUInt16LittleEndian(small.AsSpan(3+index*2,2),0xf007);
        BinaryPrimitives.WriteUInt32LittleEndian(wide.AsSpan(3+index*4,4),0xf0000007);
        BinaryPrimitives.WriteUInt32LittleEndian(legacy.AsSpan(2+index*4,4),0xf0000007);
        var box=new VoxelBox(1,1,1,2,2,18);var safe=new VoxelBox(0,0,0,1,1,1);
        Assert.Equal(1,new OctreeSpan(legacy).CountMatches(safe,VoxelFilter.Any));
        Assert.Equal(1,new OctreeSpan<ushort>(small).CountMatches(safe,VoxelFilter<ushort>.Any));
        Assert.Equal(1,new OctreeSpan<uint>(wide).CountMatches(safe,VoxelFilter<uint>.Any));
        var otherRow=new VoxelBox(0,0,1,1,1,18);var otherSlab=new VoxelBox(0,0,0,1,32,32);
        Assert.Equal(17,new OctreeSpan(legacy).CountMatches(otherRow,VoxelFilter.Any));
        Assert.Equal(17,new OctreeSpan<ushort>(small).CountMatches(otherRow,VoxelFilter<ushort>.Any));
        Assert.Equal(17,new OctreeSpan<uint>(wide).CountMatches(otherRow,VoxelFilter<uint>.Any));
        Assert.Equal(1024,new OctreeSpan(legacy).CountMatches(otherSlab,VoxelFilter.Any));
        Assert.Equal(1024,new OctreeSpan<ushort>(small).CountMatches(otherSlab,VoxelFilter<ushort>.Any));
        Assert.Equal(1024,new OctreeSpan<uint>(wide).CountMatches(otherSlab,VoxelFilter<uint>.Any));
        Assert.Throws<FormatException>(()=>new OctreeSpan(legacy).CountMatches(box,VoxelFilter.EqualTo(uint.MaxValue)));
        Assert.Throws<FormatException>(()=>new OctreeSpan<ushort>(small).CountMatches(box,VoxelFilter<ushort>.EqualTo(65535)));
        Assert.Throws<FormatException>(()=>new OctreeSpan<uint>(wide).CountMatches(box,VoxelFilter<uint>.EqualTo(uint.MaxValue)));
        if(z!=1){
            Assert.True(new OctreeSpan(legacy).Any(box,VoxelFilter.Any));
            Assert.True(new OctreeSpan<ushort>(small).Any(box,VoxelFilter<ushort>.Any));
        }
    }
}
