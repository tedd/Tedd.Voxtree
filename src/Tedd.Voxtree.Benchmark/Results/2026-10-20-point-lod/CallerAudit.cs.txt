    .Select(f => (OpCode)f.GetValue(null)!).ToDictionary(c => unchecked((ushort)c.Value));

CallerIdentity Audit(MethodInfo method)
{
    var body = method.GetMethodBody() ?? throw new InvalidOperationException("Missing IL");
    var bytes = body.GetILAsByteArray()!;
    var instructions = new List<object>();
    var module = method.Module;
    int cursor = 0;
    while (cursor < bytes.Length)
    {
        int offset = cursor;
        ushort raw = bytes[cursor++];
        if (raw == 0xfe) { Require(1); raw = (ushort)(0xfe00 | bytes[cursor++]); }
        if (!codes.TryGetValue(raw, out var code)) throw new InvalidDataException("Unknown opcode");
        object operand;
        int width;
        switch (code.OperandType)
        {
            case OperandType.InlineNone: width = 0; break;
            case OperandType.ShortInlineBrTarget: case OperandType.ShortInlineI: case OperandType.ShortInlineVar: width = 1; break;
            case OperandType.InlineVar: width = 2; break;
            case OperandType.InlineI8: case OperandType.InlineR: width = 8; break;
            case OperandType.InlineSwitch:
                Require(4); int count = BitConverter.ToInt32(bytes, cursor);
                if (count < 0) throw new InvalidDataException("Negative switch size");
                width = checked(4 + count * 4); break;
            case OperandType.InlineBrTarget: case OperandType.InlineI: case OperandType.ShortInlineR:
            case OperandType.InlineMethod: case OperandType.InlineField: case OperandType.InlineType:
            case OperandType.InlineTok: case OperandType.InlineSig: case OperandType.InlineString: width = 4; break;
            default: throw new NotSupportedException("Unsupported operand type " + code.OperandType);
        }
        Require(width);
        if (code.OperandType is OperandType.InlineMethod or OperandType.InlineField or OperandType.InlineType or OperandType.InlineTok)
            operand = Member(module.ResolveMember(BitConverter.ToInt32(bytes, cursor), method.DeclaringType!.GetGenericArguments(), method.GetGenericArguments())!);
        else if (code.OperandType == OperandType.InlineString)
            operand = module.ResolveString(BitConverter.ToInt32(bytes, cursor));
        else if (code.OperandType == OperandType.InlineSig)
            throw new NotSupportedException("Standalone call signatures require a separate semantic decoder");
        else operand = Convert.ToHexString(bytes.AsSpan(cursor, width));
        cursor += width;
        instructions.Add(new { offset, opcode = code.Name, operand });
    }
    var canonical = new { signature = Member(method), body.InitLocals, body.MaxStackSize,
        locals = body.LocalVariables.Select(v => new { type = TypeName(v.LocalType), v.IsPinned }),
        exceptions = body.ExceptionHandlingClauses.Select(c => new { flags = c.Flags.ToString(), c.TryOffset, c.TryLength, c.HandlerOffset, c.HandlerLength,
            catchType = c.Flags == ExceptionHandlingClauseOptions.Clause ? TypeName(c.CatchType!) : null,
            filter = c.Flags == ExceptionHandlingClauseOptions.Filter ? c.FilterOffset : -1 }), instructions };
    return new(method.DeclaringType!.FullName + "." + method.Name, Hash(bytes),
        Hash(Encoding.UTF8.GetBytes(JsonSerializer.Serialize(canonical))), canonical, method.GetMethodImplementationFlags().ToString());
    void Require(int size) { if (size < 0 || cursor > bytes.Length - size) throw new InvalidDataException("Truncated IL operand"); }
}

static string Member(MemberInfo member) => member switch
{
    MethodBase m => TypeName(m.DeclaringType!) + "::" + m.Name + "<" + string.Join(',', (m is MethodInfo generic ? generic.GetGenericArguments() : Type.EmptyTypes).Select(TypeName)) + ">(" +
        string.Join(',', m.GetParameters().Select(p => TypeName(p.ParameterType))) + "):" + (m is MethodInfo info ? TypeName(info.ReturnType) : "void"),
    FieldInfo f => TypeName(f.DeclaringType!) + "::" + f.Name + ":" + TypeName(f.FieldType),
    Type t => TypeName(t),
    _ => throw new NotSupportedException("Unsupported metadata operand " + member.MemberType)
};
static string TypeName(Type type) => type.IsByRef ? TypeName(type.GetElementType()!) + "&" : type.IsPointer ? TypeName(type.GetElementType()!) + "*" :
    type.IsArray ? TypeName(type.GetElementType()!) + "[" + new string(',', type.GetArrayRank() - 1) + "]" :
    type.IsGenericType ? type.GetGenericTypeDefinition().FullName + "<" + string.Join(',', type.GetGenericArguments().Select(TypeName)) + ">" : type.FullName ?? type.Name;
static string Hash(byte[] bytes) => Convert.ToHexString(SHA256.HashData(bytes));

static MethodInfo Fixture(bool reversed, bool changedCall, int constant, bool noInlining = false)
{
    var assembly = AssemblyBuilder.DefineDynamicAssembly(new AssemblyName("AuditFalsifier" + Guid.NewGuid().ToString("N")), AssemblyBuilderAccess.RunAndCollect);
    var type = assembly.DefineDynamicModule("fixture").DefineType("CallerAuditFixture", TypeAttributes.Public);
    MethodBuilder Api(string name)
    {
        var method = type.DefineMethod(name, MethodAttributes.Public | MethodAttributes.Static, typeof(int), [typeof(int)]);
        var code = method.GetILGenerator();code.Emit(OpCodes.Ldarg_0);code.Emit(OpCodes.Ret);return method;
    }
    MethodBuilder first, second;
    if (reversed) { second = Api("Second");first = Api("First"); }
    else { first = Api("First");second = Api("Second"); }
    var kernel = type.DefineMethod("AfterBatch", MethodAttributes.Public | MethodAttributes.Static, typeof(int), [typeof(int)]);
    if (noInlining) kernel.SetImplementationFlags(MethodImplAttributes.NoInlining);
    var il = kernel.GetILGenerator();il.Emit(OpCodes.Ldarg_0);il.Emit(OpCodes.Call, changedCall ? second : first);il.Emit(OpCodes.Ldc_I4, constant);il.Emit(OpCodes.Add);il.Emit(OpCodes.Ret);
    return type.CreateType()!.GetMethod("AfterBatch")!;
}
record CallerIdentity(string method, string rawIlSha256, string resolvedIlSha256, object canonical, string implementationFlags);
