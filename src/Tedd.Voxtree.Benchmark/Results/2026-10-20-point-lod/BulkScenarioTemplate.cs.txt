    bool baseline,initialized,beforePrimed,afterPrimed;
    long sum,completed,readFold;
    int zeros;
    public @CLASS@(string pattern,string op)
    {
        this.pattern=pattern;this.op=op;
        var raw=ForcecraftCorpus.Load(pattern);
        values=raw.Select(v=>(@TYPE@)v).ToArray();lods=ForcecraftCorpus.Lods(raw);
        zeros=values.Count(v=>v==0);
        for(int i=0;i<32768;i++)readFold=unchecked(readFold+(long)values[i]+((long)lods[i]<<32));
    }
    public ScenarioDescriptor Describe()=>new($"bulk-@WIDTH@-L5-{pattern}-{op}","13",

    {

        originalBefore=before.Data.ToArray();originalAfter=after.Data.ToArray();

    }
    void CheckTrees()
    {
        for(int i=0;i<32768;i++)

            int x=i/1024,y=i/32%32,z=i%32;

        }

    }

        if(!FixtureMode.Validation && !(baseline?beforePrimed:afterPrimed))

        }
        return new(sum==unchecked(expected*completed));
    }
