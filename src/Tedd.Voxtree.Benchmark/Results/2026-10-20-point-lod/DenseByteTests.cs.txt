using System.Runtime.InteropServices;
using Xunit;

namespace Tedd.Voxtree.Tests;

public class DenseByteTests
{
    private static uint[] Values() => Enumerable.Range(0, 512).Select(i => (uint)(i & 255)).ToArray();

    [Fact]
    public void AllByteValuesRoundTripThroughOwnedBorrowedMortonAndCompiledConsumers()
    {
        var values = Values();
        var tree = new Octree(3, values);
        Assert.Equal(515, tree.EncodedLength);
        Assert.Equal(3, tree.Data.Span[1] >> 6);
        Assert.Equal(3, tree.Data.Span[1] & 3);
        Assert.True(tree.AsSpan().IsWellFormed());
        Assert.Equal(tree.EncodedLength, Octree.GetRequiredSize(values, 3));
        var encoded = new byte[tree.EncodedLength];
        Assert.True(Octree.TryBuild(values, 3, encoded, out var written));
        Assert.Equal(encoded.Length, written);
        Assert.Equal(tree.Data.ToArray(), encoded);
        var shortBuffer = Enumerable.Repeat((byte)0xa5, encoded.Length - 1).ToArray();
        Assert.False(Octree.TryBuild(values, 3, shortBuffer, out written));
        Assert.Equal(0, written);
        Assert.All(shortBuffer, value => Assert.Equal((byte)0xa5, value));
        var lookup = new OctreeSpan(encoded).CreateLookup();
        var restored = Octree.FromEncoded(encoded);
        var morton = new uint[values.Length];
        DenseVoxel.Convert(values, morton, 3, DenseVoxelLayout.Linear, DenseVoxelLayout.Morton);
        var fromMorton = new Octree(3);
        fromMorton.Build(morton, DenseVoxelLayout.Morton);
        Assert.Equal(encoded, fromMorton.Data.ToArray());
        var copy = new uint[values.Length];
        tree.CopyTo(copy); Assert.Equal(values, copy);
        tree.CopyTo(copy, DenseVoxelLayout.Morton); Assert.Equal(morton, copy);
        for (int i = 0; i < values.Length; i++)
        {
            int x = i / 64, y = i / 8 % 8, z = i % 8;
            Assert.Equal(values[i], tree.Get(x, y, z, out var lod)); Assert.Equal(0, lod);
            Assert.Equal(values[i], restored.Get(x, y, z));
            Assert.Equal(values[i], lookup.Get(x, y, z, out lod)); Assert.Equal(0, lod);
        }
        encoded[3 + 83] = 251;
        Assert.Equal(251u, new OctreeSpan(encoded).Get(1, 2, 3));
        Assert.Equal(values[83], lookup.Get(1, 2, 3));
    }

    [Fact]
    public void EveryQueryKindAgreesWithIndependentDenseCoordinatesAndFilters()
    {
        var values = Values();
        var tree = new Octree(3, values);
        var filters = new[] { VoxelFilter.Any, VoxelFilter.NonZero, VoxelFilter.EqualTo(127), VoxelFilter.NotEqualTo(128), VoxelFilter.Masked(0xf0, 0x80), VoxelFilter.Masked(0x100, 0x100) };
        foreach (var box in new[] { new VoxelBox(0, 0, 0, 8, 8, 8), new VoxelBox(1, 2, 3, 7, 6, 8) })
        foreach (var filter in filters)
        {
            var expected = new List<(int X, int Y, int Z, uint Value)>();
            var region = new List<uint>();
            for (int x = box.MinX; x < box.MaxX; x++) for (int y = box.MinY; y < box.MaxY; y++) for (int z = box.MinZ; z < box.MaxZ; z++)
            {
                uint value = values[(x * 8 + y) * 8 + z]; region.Add(value);
                if (filter.Matches(value)) expected.Add((x, y, z, value));
            }
            Assert.Equal(expected.Count, tree.CountMatches(box, filter));
            Assert.Equal(expected.Count != 0, tree.Any(box, filter));
            var hits = new VoxelHit[expected.Count];
            Assert.True(tree.Query(box, filter, hits, out var written)); Assert.Equal(expected.Count, written);
            var actual = hits.Select(h => (h.X, h.Y, h.Z, h.Value)).OrderBy(h => h.X).ThenBy(h => h.Y).ThenBy(h => h.Z).ToArray();
            Assert.Equal(expected.ToArray(), actual);
            var copy = new uint[box.Count];tree.CopyRegionTo(box, copy);Assert.Equal(region.ToArray(), copy);
        }
        foreach (var filter in filters)
        {
            var expected = Enumerable.Range(0, values.Length).Select(i => new { X = i / 64, Y = i / 8 % 8, Z = i % 8, Value = values[i] })
                .Where(p => p.X * p.X + p.Y * p.Y + p.Z * p.Z <= 9 && filter.Matches(p.Value))
                .OrderBy(p => p.X * p.X + p.Y * p.Y + p.Z * p.Z).ThenBy(p => p.X).ThenBy(p => p.Y).ThenBy(p => p.Z).ToArray();
            Assert.Equal(expected.Length != 0, tree.TryFindNearest(0, 0, 0, 3, filter, out var hit));
            if (expected.Length != 0) Assert.Equal((expected[0].X, expected[0].Y, expected[0].Z, expected[0].Value), (hit.X, hit.Y, hit.Z, hit.Value));
        }
    }

    [Fact]
    public void NewHeadersRejectTruncationWrongTypesKindsVersionsAndExtraBytes()
    {
        var original = new Octree(3, Values()).Data.ToArray();
        var malformed = new List<byte[]> { original[..0], original[..1], original[..2], original[..3], original[..^1], original.Concat(new byte[] { 0 }).ToArray() };
        foreach (var change in new[] { (0, 0), (1, 0), (1, 0xce), (1, 0xc3), (1, 0xff), (2, 1), (2, 3) })
        {
            var copy = original.ToArray();copy[change.Item1] = (byte)change.Item2;malformed.Add(copy);
        }
        foreach (var bytes in malformed)
        {
            Assert.False(OctreeSpan.TryCreate(bytes, out _));
            Assert.Throws<FormatException>(() => Construct(bytes));
            Assert.Throws<FormatException>(() => Octree.FromEncoded(bytes));
        }
        var zeros = original.ToArray();Array.Clear(zeros, 3, zeros.Length - 3);
        var zeroTree = Octree.FromEncoded(zeros);
        Assert.Equal(512, zeroTree.CountMatches(new VoxelBox(0, 0, 0, 8, 8, 8), VoxelFilter.EqualTo(0)));
    }

    [Fact]
    public void WidthAndLodBoundariesRejectIneligibleByteEncoding()
    {
        foreach (uint maximum in new uint[] { 127, 128, 255, 256 })
        {
            var values = Values();values[123] = maximum;
            var tree = new Octree(3, values);
            Assert.Equal(maximum <= 255 ? 3 : 0, tree.Data.Span[1] & 3);
            Assert.Equal(maximum, tree.Get(1, 7, 3));
        }
        var mixed = Values();
        for (int x = 0; x < 2; x++) for (int y = 0; y < 2; y++) for (int z = 0; z < 2; z++) mixed[(x * 8 + y) * 8 + z] = 7;
        var coarse = new Octree(3, mixed);
        Assert.Equal(0, coarse.Data.Span[1] & 3);
        Assert.Equal(7u, coarse.Get(0, 0, 0, out var lod));Assert.Equal(1, lod);
        Assert.Throws<ArgumentException>(() => OverlappingBuild(Values()));
    }
    private static void Construct(byte[] bytes) => _ = new OctreeSpan(bytes);
    private static void OverlappingBuild(uint[] values) => Octree.TryBuild(values, 3, MemoryMarshal.AsBytes(values.AsSpan()), out _);
}
