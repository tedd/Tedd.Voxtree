using System;
using System.Buffers.Binary;
using System.Runtime.CompilerServices;
using System.Runtime.InteropServices;

namespace Tedd.Voxtree;

// Isolated P15 prototype; not a production wire-format commitment.
internal static class PackedLeafEncoding<T> where T : unmanaged, IEquatable<T>
{
    private static readonly int Width = Unsafe.SizeOf<T>();
    private static readonly int Bits = Width * 8 - 4;
    private static readonly uint Mask = Width == 2 ? 0xfffu : 0xfffffffu;

    internal static bool Select(ReadOnlySpan<T> source, int levels, int treeBytes, int denseBytes)
    {
        if (Width is not (2 or 4) || levels is < 3 or > 6 || treeBytes * 8L < denseBytes) return false;
        if (Width == 2)
        {
            var values = MemoryMarshal.Cast<T, ushort>(source);
#if NET10_0_OR_GREATER
            return values.IndexOfAnyExceptInRange((ushort)0, (ushort)0xfff) < 0;
#else
            foreach (var value in values) if (value > 0xfff) return false;
#endif
        }
        else
        {
            var values = MemoryMarshal.Cast<T, uint>(source);
#if NET10_0_OR_GREATER
            return values.IndexOfAnyExceptInRange(0u, 0xfffffffu) < 0;
#else
            foreach (var value in values) if (value > 0xfffffff) return false;
#endif
        }
        return true;
    }

    internal static bool TryReadHeader(ReadOnlySpan<byte> data, int header, out int levels)
    {
        levels = 0;
        if (Width is not (2 or 4) || data.Length < header || data[0] != 0x4f ||
            data[1] >> 6 != 3 || (data[1] & 3) != 3 ||
            (header == 3 && data[2] != (Width == 2 ? 1 : 2))) return false;
        levels = (data[1] >> 2) & 15;
        return levels <= Octree.MaxLevels &&
            data.Length == header + (long)OctreeCodec.GetVoxelCount(levels) * Width;
    }

    internal static void Write(ReadOnlySpan<T> source, int levels, DenseVoxelLayout layout,
        Span<byte> destination, int header)
    {
        var count = OctreeCodec.GetVoxelCount(levels);
        destination = destination[..checked(header + count * Width)];
        destination[0] = 0x4f;
        destination[1] = (byte)(0xc3 | (levels << 2));
        if (header == 3) destination[2] = (byte)(Width == 2 ? 1 : 2);
        WriteCube(source, levels, layout, destination[header..], 0, 0, 0, levels);
    }

    private static void WriteCube(ReadOnlySpan<T> source, int levels, DenseVoxelLayout layout,
        Span<byte> body, int x, int y, int z, int lod)
    {
        var side = 1 << levels;
        var size = 1 << lod;
        if (Uniform(source, side, layout, x, y, z, size))
        {
            var value = Raw(source[DenseVoxel.Index(x, y, z, side, layout)]);
            var packed = value | ((uint)lod << Bits);
            for (var xx = x; xx < x + size; xx++)
            for (var yy = y; yy < y + size; yy++)
            {
                var row = body.Slice(((xx * side + yy) * side + z) * Width, size * Width);
                if (BitConverter.IsLittleEndian)
                {
                    if (Width == 2) MemoryMarshal.Cast<byte, ushort>(row).Fill((ushort)packed);
                    else MemoryMarshal.Cast<byte, uint>(row).Fill(packed);
                }
                else for (var zz = 0; zz < size; zz++) WriteWord(row.Slice(zz * Width, Width), packed);
            }
            return;
        }
        var half = size >> 1;
        for (var child = 0; child < 8; child++)
            WriteCube(source, levels, layout, body, x + ((child >> 2) & 1) * half,
                y + ((child >> 1) & 1) * half, z + (child & 1) * half, lod - 1);
    }

    private static bool Uniform(ReadOnlySpan<T> source, int side, DenseVoxelLayout layout,
        int x, int y, int z, int size)
    {
        var start = DenseVoxel.Index(x, y, z, side, layout);
        var value = source[start];
        if (layout == DenseVoxelLayout.Morton) return Equal(source.Slice(start, size * size * size), value);
        for (var xx = x; xx < x + size; xx++)
        for (var yy = y; yy < y + size; yy++)
            if (!Equal(source.Slice((xx * side + yy) * side + z, size), value)) return false;
        return true;
    }

    private static bool Equal(ReadOnlySpan<T> values, T value)
    {
#if NET10_0_OR_GREATER
        return values.IndexOfAnyExcept(value) < 0;
#else
        foreach (var cell in values) if (!cell.Equals(value)) return false;
        return true;
#endif
    }

    private static uint Raw(T value) => Width == 2 ? Unsafe.As<T, ushort>(ref value) : Unsafe.As<T, uint>(ref value);
    private static T Value(uint value)
    {
        T result = default;
        if (Width == 2) Unsafe.As<T, ushort>(ref result) = (ushort)value;
        else Unsafe.As<T, uint>(ref result) = value;
        return result;
    }
    private static uint ReadWord(ReadOnlySpan<byte> value) => Width == 2
        ? BinaryPrimitives.ReadUInt16LittleEndian(value) : BinaryPrimitives.ReadUInt32LittleEndian(value);
    private static void WriteWord(Span<byte> destination, uint value)
    {
        if (Width == 2) BinaryPrimitives.WriteUInt16LittleEndian(destination, (ushort)value);
        else BinaryPrimitives.WriteUInt32LittleEndian(destination, value);
    }

    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    internal static bool TryGet(ReadOnlySpan<byte> data, int header, int levels,
        int x, int y, int z, out T value, out int lod)
    {
        value = default;
        lod = 0;
        if (Width is not (2 or 4)) return false;
        var index = ((x << levels) + y) * (1 << levels) + z;
        var offset = header + index * Width;
        if ((uint)offset > (uint)data.Length || data.Length - offset < Width) return false;
        var word = ReadWord(data.Slice(offset, Width));
        var stored = (int)(word >> Bits);
        if (stored > levels) return false;
        value = Value(word & Mask);
        lod = stored;
        return true;
    }

    internal static bool IsWellFormed(ReadOnlySpan<byte> data, int header, int levels) =>
        TryReadHeader(data, header, out var actual) && actual == levels &&
        ValidateCube(data[header..], 1 << levels, 0, 0, 0, levels);

    private static bool ValidateCube(ReadOnlySpan<byte> body, int side, int x, int y, int z, int level)
    {
        var word = ReadWord(body.Slice(((x * side + y) * side + z) * Width, Width));
        var lod = (int)(word >> Bits);
        if (lod > level) return false;
        if (lod == level)
        {
            var size = 1 << level;
            for (var xx = x; xx < x + size; xx++)
            for (var yy = y; yy < y + size; yy++)
            for (var zz = z; zz < z + size; zz++)
                if (ReadWord(body.Slice(((xx * side + yy) * side + zz) * Width, Width)) != word) return false;
            return true;
        }
        var half = 1 << (level - 1);
        for (var child = 0; child < 8; child++)
            if (!ValidateCube(body, side, x + ((child >> 2) & 1) * half,
                y + ((child >> 1) & 1) * half, z + (child & 1) * half, level - 1)) return false;
        return true;
    }

    internal static bool IsEmpty(ReadOnlySpan<byte> data, int header)
    {
        for (var cursor = header; cursor < data.Length; cursor += Width)
            if ((ReadWord(data.Slice(cursor, Width)) & Mask) != 0) return false;
        return true;
    }

    internal static void CopyTo(ReadOnlySpan<byte> data, int header, int levels,
        Span<T> destination, DenseVoxelLayout layout)
    {
        var side = 1 << levels;
        var cursor = header;
        for (var x = 0; x < side; x++)
        for (var y = 0; y < side; y++)
        for (var z = 0; z < side; z++, cursor += Width)
            destination[DenseVoxel.Index(x, y, z, side, layout)] = Value(ReadWord(data.Slice(cursor, Width)) & Mask);
    }
}
